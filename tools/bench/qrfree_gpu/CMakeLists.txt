cmake_minimum_required(VERSION 3.24)
project(QRFreeCost LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(qrfree_cost qrfree_cost.cu)
target_compile_features(qrfree_cost PRIVATE cxx_std_17 cuda_std_17)
target_link_libraries(qrfree_cost PRIVATE CUDA::cudart CUDA::cublas CUDA::cusparse CUDA::cusolver nlohmann_json::nlohmann_json Eigen3::Eigen)
target_compile_options(qrfree_cost PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;--expt-relaxed-constexpr;-Xcompiler=/utf-8>)
set_target_properties(qrfree_cost PROPERTIES CUDA_ARCHITECTURES 86)
