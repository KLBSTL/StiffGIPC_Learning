cmake_minimum_required(VERSION 3.18)
project(MASReplay LANGUAGES CXX CUDA)
find_package(Eigen3 REQUIRED)
find_package(CUDAToolkit REQUIRED)
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../stiff_perf_v37/StiffGIPC")
add_executable(mas_replay replay.cu "${UPSTREAM}/linear_system/utils/spmv.cu")
target_compile_features(mas_replay PRIVATE cuda_std_17)
target_compile_options(mas_replay PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--default-stream=per-thread;--use_fast_math;--extended-lambda;--expt-relaxed-constexpr>)
target_include_directories(mas_replay PRIVATE "${UPSTREAM}")
target_link_libraries(mas_replay PRIVATE Eigen3::Eigen CUDA::cudart CUDA::cublas CUDA::cusparse)
set_target_properties(mas_replay PROPERTIES CUDA_ARCHITECTURES 89)
enable_testing()
add_test(NAME cholesky_factor_action_graph COMMAND mas_replay --selftest "${CMAKE_CURRENT_BINARY_DIR}/factor_fixture.json")
