cmake_minimum_required(VERSION 3.22)
project(StiffGIPCObserved LANGUAGES CXX CUDA)
get_filename_component(BASE "${CMAKE_CURRENT_SOURCE_DIR}/../stiff_base" ABSOLUTE)
add_subdirectory("${BASE}" "${CMAKE_CURRENT_BINARY_DIR}/inherited-base")
get_target_property(inputs gipc SOURCES)
list(REMOVE_ITEM inputs "${BASE}/StiffGIPC/app/gl_main.cu")
list(APPEND inputs "${CMAKE_CURRENT_SOURCE_DIR}/StiffGIPC/app/gl_main.cu")
set_property(TARGET gipc PROPERTY SOURCES "${inputs}")
target_include_directories(gipc BEFORE PRIVATE "${BASE}/StiffGIPC/app")
set_target_properties(gipc PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
